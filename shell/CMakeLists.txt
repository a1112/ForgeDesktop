cmake_minimum_required(VERSION 3.25)
project(ForgeShell VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Qml Quick DBus Test)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-unix-2.0)
add_library(shell-model src/model.cpp src/model.h)
target_include_directories(shell-model PUBLIC src)
target_link_libraries(shell-model PUBLIC Qt6::Core Qt6::Gui PkgConfig::GIO)
target_compile_options(shell-model PRIVATE -Wall -Wextra -Werror)
add_executable(model-test tests/model_test.cpp)
target_link_libraries(model-test PRIVATE shell-model Qt6::Test)
add_library(notification-client src/notifications.cpp src/notifications.h)
target_include_directories(notification-client PUBLIC src)
target_link_libraries(notification-client PUBLIC Qt6::Core Qt6::DBus)
target_compile_options(notification-client PRIVATE -Wall -Wextra -Werror)
add_library(tray-client src/tray.cpp src/tray.h)
target_include_directories(tray-client PUBLIC src)
target_link_libraries(tray-client PUBLIC Qt6::Core Qt6::DBus)
add_executable(notifications-test tests/notifications_test.cpp)
target_link_libraries(notifications-test PRIVATE notification-client Qt6::Test)
add_executable(tray-test tests/tray_test.cpp)
target_link_libraries(tray-test PRIVATE tray-client Qt6::Test)
enable_testing()
add_test(NAME model COMMAND model-test)
add_test(NAME notifications COMMAND /usr/bin/dbus-run-session -- $<TARGET_FILE:notifications-test>)
add_test(NAME tray COMMAND /usr/bin/dbus-run-session -- $<TARGET_FILE:tray-test>)
add_executable(forge-shell src/main.cpp)
target_link_libraries(forge-shell PRIVATE shell-model notification-client tray-client Qt6::Qml Qt6::Quick)
target_compile_options(forge-shell PRIVATE -Wall -Wextra -Werror)
qt_add_resources(forge-shell "desktop" PREFIX "/" FILES qml/Desktop.qml qml/Action.qml)
install(TARGETS forge-shell RUNTIME DESTINATION libexec/forge-desktop)
